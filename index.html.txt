    const POINTAGE_MINUTES = 30;
    const LIME = "#D7FF3F";
    const INK = "#14151A";
    const BG = "#F5F5F1";
    const CARD = "#FFFFFF";
    const BORDER = "#E7E7E2";
    const MUTED = "#8A8D93";
    const GREEN = "#20C171";

    const AGENT_COLORS = ["#D7FF3F", "#5CC8FF", "#FF9F5C", "#C89CFF", "#FF5C9E", "#5CFFB0"];
    const DAY_LABELS = ["Lun", "Mar", "Mer", "Jeu", "Ven", "Sam", "Dim"];

    function pad(n) { return String(n).padStart(2, "0"); }
    function toDateStr(d) { return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`; }
    function isSameDate(a, b) { return toDateStr(a) === toDateStr(b); }
    function getMonday(d) {
      const date = new Date(d);
      const day = date.getDay();
      const diff = day === 0 ? -6 : 1 - day;
      date.setDate(date.getDate() + diff);
      date.setHours(0, 0, 0, 0);
      return date;
    }
    function addDays(d, n) {
      const date = new Date(d);
      date.setDate(date.getDate() + n);
      return date;
    }
    function getMonthDates(d) {
      const year = d.getFullYear(), month = d.getMonth();
      const days = new Date(year, month + 1, 0).getDate();
      return Array.from({ length: days }, (_, i) => new Date(year, month, i + 1));
    }
    function slotMinutes(slot) {
      if (!slot.start || !slot.end) return 0;
      const [sh, sm] = slot.start.split(":").map(Number);
      const [eh, em] = slot.end.split(":").map(Number);
      let start = sh * 60 + sm;
      let end = eh * 60 + em;
      if (end <= start) end += 24 * 60;
      return (end - start) + POINTAGE_MINUTES;
    }
    function minutesToHM(min) {
      const h = Math.floor(min / 60);
      const m = min % 60;
      return `${h}h${pad(m)}`;
    }

      const selectedDateStr = toDateStr(selectedDate);

        const { data, error } = await supabase.from('agents').select('*').order('id', { ascending: true });
        if (!error && data) setAgents(data);
      };

      useEffect(() => { loadAgents(); }, []);

        if (setLoadingFn) setLoadingFn(true);

        const { data, error } = await supabase
          .from('slots')
          .select('*')
          .in('date_str', dateStrings);

        const entries = {};
        dateStrings.forEach(ds => { entries[ds] = []; });

        if (!error && data) {
          data.forEach(slot => {
            if (entries[slot.date_str]) {
              entries[slot.date_str].push(slot);
            }
          });
        }

        setSlotsByDate((prev) => ({ ...prev, ...entries }));
        if (setLoadingFn) setLoadingFn(false);
      }, []);

      useEffect(() => { loadDates(weekDates, setLoading); }, [weekStart]);

        if (activePanel === "totals" && totalsScope === "month") {
          loadDates(getMonthDates(totalsMonth), setTotalsLoading);
        }
      }, [activePanel, totalsScope, totalsMonth]);

      async function addAgent() {
        const name = newAgentName.trim();
        if (!name) return;
        const color = AGENT_COLORS[agents.length % AGENT_COLORS.length];

        const { data, error } = await supabase.from('agents').insert([{ name, color }]).select();
        if (error) {
          setError("Erreur lors de l'ajout de l'agent.");
        } else if (data) {
          setAgents([...agents, data[0]]);
          setNewAgentName("");
        }
      }

      async function deleteAgent(id) {
        const { error } = await supabase.from('agents').delete().eq('id', id);
        if (!error) {
          setAgents(agents.filter((a) => a.id !== id));
        } else {
          setError("Impossible de supprimer cet agent.");
        }
      }

        if (!newSlot.start || !newSlot.end) return;

        const slotToInsert = {
          date_str: selectedDateStr,
          start: newSlot.start,
          end: newSlot.end,
          club: newSlot.club.trim(),
          agent_id: null
        };

        const { data, error } = await supabase.from('slots').insert([slotToInsert]).select();

        if (error) {
          setError("Erreur lors de la création du créneau.");
        } else if (data) {
          const updated = [...(slotsByDate[selectedDateStr] || []), data[0]];
          updated.sort((a, b) => a.start.localeCompare(b.start));
          setSlotsByDate(prev => ({ ...prev, [selectedDateStr]: updated }));
          setNewSlot({ start: "09:00", end: "13:00", club: "" });
          setShowAddSlot(false);
        }
      }

        const targetAgentId = agentId ? parseInt(agentId, 10) : null;

        if (!error) {
          const updated = (slotsByDate[selectedDateStr] || []).map((s) =>
            s.id === slotId ? { ...s, agent_id: targetAgentId } : s
          );
          setSlotsByDate(prev => ({ ...prev, [selectedDateStr]: updated }));
        } else {
          setError("Erreur lors de l'assignation de l'agent.");
        }
      }

      async function deleteSlot(slotId) {
        const { error } = await supabase.from('slots').delete().eq('id', slotId);
        if (!error) {
          const updated = (slotsByDate[selectedDateStr] || []).filter((s) => s.id !== slotId);
          setSlotsByDate(prev => ({ ...prev, [selectedDateStr]: updated }));
        } else {
          setError("Erreur lors de la suppression du créneau.");
        }
      }

        const slots = slotsByDate[dateStr] || [];
        const filled = slots.filter((s) => s.agent_id).length;
        return { filled, total: slots.length };
      }

            if (s.agent_id) {
              totals[s.agent_id] = (totals[s.agent_id] || 0) + slotMinutes(s);
            } else {
              openCount += 1;
            }
          });
        });
        return { totals, openCount };
      }

      return (
        <div style={{ minHeight: "100vh", background: BG, fontFamily: "'Inter', system-ui, sans-serif", color: INK, paddingBottom: 40 }}>

              <span style={{ fontSize: 13, fontWeight: 700, color: INK }}>{weekLabel}</span>
              <button onClick={() => setWeekStart(addDays(weekStart, 7))} style={{ background: CARD, border: `2px solid ${BORDER}`, borderRadius: 8, padding: "6px 12px", fontWeight: "bold" }}>›</button>
            </div>

                {daySlots.map((slot) => {
                  const agent = agentById(slot.agent_id);

                    <div key={slot.id} style={{ background: CARD, border: `2px solid ${agent ? BORDER : CORAL}`, borderRadius: 14, padding: 12 }}>
                      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "flex-start" }}>
                        <div>
                          <div style={{ fontSize: 16, fontWeight: 700 }}>{slot.start} – {slot.end}</div>
                          {slot.club && (
                            <div style={{ fontSize: 12.5, color: MUTED, display: "flex", alignItems: "center", gap: 4, marginTop: 4, fontWeight: 600 }}>
                              📍 {slot.club}
                            </div>
                          )}
                        </div>
                        <button onClick={() => deleteSlot(slot.id)} style={{ background: "none", border: "none", padding: 4, color: MUTED, cursor: "pointer" }}>
                          🗑️
                        </button>
                      </div>
                      <div style={{ marginTop: 10, display: "flex", alignItems: "center", gap: 8 }}>
                        {agent && <span style={{ width: 10, height: 10, borderRadius: "50%", background: agent.color, flexShrink: 0 }} />}
                        <select
                          value={slot.agent_id || ""}
                          onChange={(e) => assignAgent(slot.id, e.target.value)}
                          style={{ flex: 1, fontWeight: 600, padding: 8, borderRadius: 8, border: `2px solid ${BORDER}` }}

                          ))}

                      </div>
                      {!agent && (
                        <div style={{ marginTop: 8, fontSize: 12, color: CORAL, fontWeight: 600, display: "flex", alignItems: "center", gap: 4 }}>
                          ⚠️ Remplaçant à trouver

                    </div>

                  <div style={{ flex: 1 }}>
                    <label style={{ fontSize: 11, color: MUTED, fontWeight: 600 }}>Début</label>
                    <input type="time" value={newSlot.start} onChange={(e) => setNewSlot({ ...newSlot, start: e.target.value })} style={{ width: "100%", display: "block", marginTop: 4, padding: 8, border: `2px solid ${BORDER}`, borderRadius: 8 }} />

                  <div style={{ flex: 1 }}>

                </div>

            ) : (
              <button
                onClick={() => setShowAddSlot(true)}
                style={{ width: "100%", marginTop: 12, background: INK, color: LIME, border: "none", borderRadius: 12, padding: "13px 0", display: "flex", alignItems: "center", justifyContent: "center", gap: 6, fontWeight: 700, fontSize: 14 }}
              >

            <div style={{ position: "fixed", inset: 0, background: "rgba(20,21,26,0.5)", zIndex: 50, display: "flex", justifyContent: "flex-end" }} onClick={() => setActivePanel(null)}>
              <div onClick={(e) => e.stopPropagation()} style={{ width: "min(340px, 88vw)", background: BG, height: "100%", padding: 18, overflowY: "auto" }}>
                <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 16 }}>
                  <h2 style={{ fontFamily: "'Archivo Black', system-ui", fontSize: 17, margin: 0 }}>Agents</h2>

                </div>

                <div style={{ display: "flex", gap: 8, marginBottom: 16 }}>
                  <input
                    type="text"
                    placeholder="Nom de l'agent"
                    value={newAgentName}
                    onChange={(e) => setNewAgentName(e.target.value)}
                    onKeyDown={(e) => e.key === "Enter" && addAgent()}
                    style={{ flex: 1, padding: 8, border: `2px solid ${BORDER}`, borderRadius: 8 }}
                  />
                  <button onClick={addAgent} style={{ background: INK, color: LIME, border: "none", borderRadius: 8, padding: "0 14px", fontWeight: 700 }}>+</button>

                {agents.length === 0 ? (
                  <div style={{ color: MUTED, fontSize: 13.5 }}>Aucun agent pour l'instant.</div>

                  <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
                    {agents.map((a) => (
                      <div key={a.id} style={{ background: CARD, border: `2px solid ${BORDER}`, borderRadius: 10, padding: "10px 12px", display: "flex", alignItems: "center", gap: 10 }}>
                        <span style={{ width: 12, height: 12, borderRadius: "50%", background: a.color, flexShrink: 0 }} />
                        <span style={{ flex: 1, fontWeight: 600, fontSize: 14 }}>{a.name}</span>
                        <button onClick={() => deleteAgent(a.id)} style={{ background: "none", border: "none", color: MUTED }}>🗑️</button>
                      </div>
                    ))}
                  </div>

              </div>
            </div>
          )}

          {/* Panneau Totaux */}
          {activePanel === "totals" && (() => {
            const dates = totalsScope === "week" ? weekDates : getMonthDates(totalsMonth);
            const { totals, openCount } = computeTotals(dates);
            const sortedAgents = [...agents].sort((a, b) => (totals[b.id] || 0) - (totals[a.id] || 0));
            const grandTotal = Object.values(totals).reduce((a, b) => a + b, 0);

                <div onClick={(e) => e.stopPropagation()} style={{ width: "min(360px, 88vw)", background: BG, height: "100%", padding: 18, overflowY: "auto" }}>
                  <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 14 }}>
                    <h2 style={{ fontFamily: "'Archivo Black', system-ui", fontSize: 17, margin: 0 }}>Totaux d'heures</h2>

                      <button
                        key={scope}

                          flex: 1, padding: "8px 0", borderRadius: 8, border: "none", fontWeight: 700, fontSize: 13,

                  )}

                  {totalsLoading ? (
                    <div style={{ color: MUTED, fontSize: 14 }}>Chargement…</div>
                  ) : (
                    <>
                      <div style={{ background: INK, borderRadius: 14, padding: "16px", marginBottom: 14, textAlign: "center" }}>
                        <div style={{ fontFamily: "'Archivo Black', system-ui", fontSize: 28, color: LIME }}>{minutesToHM(grandTotal)}</div>
                        <div style={{ fontSize: 11.5, color: "rgba(255,255,255,0.6)", marginTop: 2 }}>heures couvertes au total</div>
                      </div>

                      {sortedAgents.length === 0 ? (
                        <div style={{ color: MUTED, fontSize: 13.5 }}>Aucun agent à afficher.</div>
                      ) : (
                        <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
                          {sortedAgents.map((a) => {
                            const mins = totals[a.id] || 0;
                            return (
                              <div key={a.id} style={{ background: CARD, border: `2px solid ${BORDER}`, borderRadius: 10, padding: "10px 12px", display: "flex", alignItems: "center", gap: 10 }}>
                                <span style={{ width: 12, height: 12, borderRadius: "50%", background: a.color, flexShrink: 0 }} />
                                <span style={{ flex: 1, fontWeight: 600, fontSize: 14 }}>{a.name}</span>
                                <span style={{ fontWeight: 700, fontSize: 14.5 }}>{minutesToHM(mins)}</span>
                              </div>
                            );
                          })}
                        </div>
                      )}

                      {openCount > 0 && (
                        <div style={{ marginTop: 12, background: "#FFF0F0", border: `1px solid ${CORAL}`, borderRadius: 10, padding: "10px 12px", display: "flex", alignItems: "center", gap: 8, fontSize: 12.5, color: INK }}>
                          ⚠️ {openCount} créneau{openCount > 1 ? "x" : ""} non assigné{openCount > 1 ? "s" : ""}
                        </div>
                      )}
                    </>
                  )}

                  <div style={{ marginTop: 16, fontSize: 11.5, color: MUTED, lineHeight: 1.5 }}>
                    Chaque créneau inclut 30 min de pointage effectuées avant le début du service.
                  </div>
                </div>
              </div>
            );
          })()}
        </div>
      );
    }

    ReactDOM.createRoot(document.getElementById("root")).render(<App />);
